57 void begin(uint8_t cols, uint8_t lines,
58 uint8_t dotsize = LCD_5x8DOTS) override {
59 _displayfunction &= ~(LCD_2LINE | LCD_5x10DOTS);
60 if (lines > 1) {
61 _displayfunction |= LCD_2LINE;
62 }
63 _numlines = lines;
64
65 if (_led_a != 0) {
66 pinModeLCD(_led_a, OUTPUT);
67 }
68
69 setRowOffsets(0x00, 0x40, 0x00 + cols, 0x40 + cols);
70
71 // for some 1 line displays you can select a 10 pixel high font
72 if ((dotsize != LCD_5x8DOTS) && (lines == 1)) {
73 _displayfunction |= LCD_5x10DOTS;
74 }
75
76 pinModeLCD(_rs_pin, OUTPUT);
77 // we can save 1 pin by not using RW. Indicate by passing 255 instead of
78 // pin#
79 if (_rw_pin != 255) {
80 pinModeLCD(_rw_pin, OUTPUT);
81 }
82 pinModeLCD(_enable_pin, OUTPUT);
83
84 // Do these once, instead of every time a character is drawn for speed
85 // reasons.
86 for (int i = 0; i < ((_displayfunction & LCD_8BITMODE) ? 8 : 4); ++i) {
87 pinModeLCD(_data_pins[i], OUTPUT);
88 }
89
90 // SEE PAGE 45/46 FOR INITIALIZATION SPECIFICATION!
91 // according to datasheet, we need at least 40 ms after power rises
92 // above 2.7 V before sending commands. Arduino can turn on way before 4.5 V
93 // so we'll wait 50
94 delayMicrosecondsLCD(50000);
95 // Now we pull both RS and R/W low to begin commands
96 digitalWriteLCD(_rs_pin, LOW);
97 digitalWriteLCD(_enable_pin, LOW);
98 if (_rw_pin != 255) {
99 digitalWriteLCD(_rw_pin, LOW);
100 }
101
102 // put the LCD into 4 bit or 8 bit mode
103 if (!(_displayfunction & LCD_8BITMODE)) {
104 // this is according to the Hitachi HD44780 datasheet
105 // figure 24, pg 46
106
107 // we start in 8bit mode, try to set 4 bit mode
108 write4bits(0x03);
109 delayMicrosecondsLCD(4500); // wait min 4.1ms
110
111 // second try
112 write4bits(0x03);
113 delayMicrosecondsLCD(4500); // wait min 4.1ms
114
115 // third go!
116 write4bits(0x03);
117 delayMicrosecondsLCD(150);
118
119 // finally, set to 4-bit interface
120 write4bits(0x02);
121 } else {
122 // this is according to the Hitachi HD44780 datasheet
123 // page 45 figure 23
124
125 // Send function set command sequence
126 command(LCD_FUNCTIONSET | _displayfunction);
127 delayMicrosecondsLCD(4500); // wait more than 4.1 ms
128
129 // second try
130 command(LCD_FUNCTIONSET | _displayfunction);
131 delayMicrosecondsLCD(150);
132
133 // third go
134 command(LCD_FUNCTIONSET | _displayfunction);
135 }
136
137 // finally, set # lines, font size, etc.
138 command(LCD_FUNCTIONSET | _displayfunction);
139
140 // turn the display on with no cursor or blinking default
141 _displaycontrol = LCD_DISPLAYON | LCD_CURSOROFF | LCD_BLINKOFF;
142 display();
143
144 // clear it off
145 clear();
146
147 // Initialize to default text direction (for romance languages)
148 _displaymode = LCD_ENTRYLEFT | LCD_ENTRYSHIFTDECREMENT;
149 // set the entry mode
150 command(LCD_ENTRYMODESET | _displaymode);
151 }
